\documentclass{article}

% Conditionally redefine Tufte-style macros for HTML output
\ifdefined\htmlversion
  \def\newthought#1{@@newthought:#1@@}
  \def\marginnote#1{@@marginnote:#1@@}
  \def\sidenote#1{@@sidenote:#1@@}
\else
  \newcommand{\newthought}[1]{\textsc{#1}}
  \newcommand{\marginnote}[1]{\marginpar{#1}}
  \newcommand{\sidenote}[1]{\footnote{#1}}
\fi


\begin{document}

\newthought{This is a new thought} that begins this section with special formatting. 
Regular text follows to show the contrast.

More content here. Let's add a \sidenote{Side note example} in the middle of the sentence.

Another paragraph with a \marginnote{Margin note example}.

\end{document}
